Standardize Numeric Predictors

Scaling is based only on the training set.

Code 
train_means <- train_data |>
  summarise(
    across(
      all_of(numeric_predictors),
      mean
    )
  )

train_sds <- train_data |>
  summarise(
    across(
      all_of(numeric_predictors),
      sd
    )
  )

train_scaled <- train_data
test_scaled <- test_data

for (x in numeric_predictors) {
  train_scaled[[x]] <-
    (train_scaled[[x]] - train_means[[x]]) /
    train_sds[[x]]

  test_scaled[[x]] <-
    (test_scaled[[x]] - train_means[[x]]) /
    train_sds[[x]]
}

Evaluation Function

Code 
evaluate_model <- function(
    model_name,
    actual,
    predicted,
    probabilities = NULL) {

  predicted <- factor(
    predicted,
    levels = levels(actual)
  )

  confusion <- table(
    Actual = actual,
    Predicted = predicted
  )

  accuracy <- sum(diag(confusion)) / sum(confusion)

  recall <- diag(confusion) / rowSums(confusion)
  precision <- diag(confusion) / colSums(confusion)

  precision[is.na(precision)] <- 0
  recall[is.na(recall)] <- 0

  f1 <- ifelse(
    precision + recall == 0,
    0,
    2 * precision * recall / (precision + recall)
  )

  macro_f1 <- mean(f1)
  balanced_accuracy <- mean(recall)

  log_loss <- NA_real_

  if (!is.null(probabilities)) {
    probabilities <- as.matrix(probabilities)

    probabilities <- probabilities[
      ,
      levels(actual),
      drop = FALSE
    ]

    actual_column <- match(
      as.character(actual),
      colnames(probabilities)
    )

    actual_probability <- probabilities[
      cbind(
        seq_along(actual),
        actual_column
      )
    ]

    actual_probability <- pmax(
      actual_probability,
      1e-15
    )

    log_loss <- -mean(log(actual_probability))
  }

  list(
    overall = tibble(
      model = model_name,
      accuracy = accuracy,
      macro_f1 = macro_f1,
      balanced_accuracy = balanced_accuracy,
      log_loss = log_loss
    ),

    class = tibble(
      model = model_name,
      class = names(recall),
      precision = as.numeric(precision),
      recall = as.numeric(recall),
      f1 = as.numeric(f1)
    ),

    confusion = as.data.frame(confusion) |>
      as_tibble() |>
      mutate(model = model_name) |>
      relocate(model)
  )
}

K-Nearest Neighbors

Code 
k_grid <- seq(
  3,
  31,
  by = 2
)

knn_cv_accuracy <- numeric(
  length(k_grid)
)

for (j in seq_along(k_grid)) {
  k <- k_grid[j]
  fold_accuracy <- numeric(5)

  for (fold in 1:5) {
    validation_rows <- which(
      fold_id == fold
    )

    cv_pred <- class::knn(
      train = train_x[-validation_rows, ],
      test = train_x[validation_rows, ],
      cl = train_y[-validation_rows],
      k = k
    )

    fold_accuracy[fold] <- mean(
      cv_pred == train_y[validation_rows]
    )
  }

  knn_cv_accuracy[j] <- mean(
    fold_accuracy
  )
}

knn_cv_results <- tibble(
  k = k_grid,
  cv_accuracy = knn_cv_accuracy
)

best_k <- knn_cv_results |>
  slice_max(
    cv_accuracy,
    n = 1,
    with_ties = FALSE
  ) |>
  pull(k)

knn_prob <- matrix(
  0,
  nrow = nrow(test_x),
  ncol = length(levels(train_y))
)

colnames(knn_prob) <- levels(train_y)

for (i in seq_len(nrow(test_x))) {
  distances <- rowSums(
    sweep(
      train_x,
      2,
      test_x[i, ],
      "-"
    )^2
  )

  neighbors <- order(
    distances
  )[1:best_k]

  neighbor_classes <- factor(
    train_y[neighbors],
    levels = levels(train_y)
  )

  knn_prob[i, ] <-
    as.numeric(
      table(neighbor_classes)
    ) / best_k
}

knn_pred <- factor(
  colnames(knn_prob)[
    max.col(
      knn_prob,
      ties.method = "first"
    )
  ],
  levels = levels(train_y)
)

results[["KNN"]] <- evaluate_model(
  paste0("KNN (k=", best_k, ")"),
  test_y,
  knn_pred,
  knn_prob
)

set.seed(131)

elastic_alpha_grid <- c(
  0.25,
  0.50,
  0.75
)

elastic_cv_results <- map_dfr(
  elastic_alpha_grid,
  function(a) {
    fit <- glmnet::cv.glmnet(
      x = train_x,
      y = train_y,
      family = "multinomial",
      alpha = a,
      foldid = fold_id,
      type.measure = "deviance"
    )

    tibble(
      alpha = a,
      lambda = fit$lambda.min,
      cv_deviance = min(fit$cvm)

best_elastic <- elastic_cv_results |>
